@ Conversiones explícitas de Punto Flotante a Entero
PROGRAMA {
    var A := 5UI;
    var B := 5.0F+0;
    var D := 15.2F+0;

    var C := A + toui(B);
    print(&C = 10&);
    print(C);
    C := toui(D + B);
    print(&C = 20&);
    print(C);
    var E := A + toui(2.5F+0);
    print(&E = 7&);
    print(E);
}
