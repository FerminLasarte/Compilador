    var A := 5UI;
    uint FUNCION(lambda B, uint C) {
        B(C);
    }

    FUNCION((uint A) {
        if(PROGRAMA.A > 3UI) {
            print(&Imprime Lambda Positivo&);
        } else {
            print(&Imprime Lambda Negativo&);

        } endif;
    }->B, 5UI->C);
}
